# FE AUTH HTTP-ONLY MANUAL SETUP
### This is a traced, manual setup of http-only authenticaion from another codebased
### Strategy here is via tracing from configs, to main.ts to app.tsx upto the lowest component involved

# CONFIG FILES
## ENV `.env`
```dotenv
    VITE_API_BASE_URL=
    # DEV ONLY
    VITE_DEV_PROXY_TARGET=http://localhost:30001
    VITE_APP_ENV=development
    VITE_APP_DSN=
```
## VITE CONFIG TS `vite.config.ts`
```ts 
    import { defineConfig, loadEnv } from 'vite'
    import react, { reactCompilerPreset } from '@vitejs/plugin-react'
    import babel from '@rolldown/plugin-babel'
    import tailwindcss from '@tailwindcss/vite'
    import path, { dirname } from 'node:path'
    import { fileURLToPath } from 'node:url'
    
    // https://vite.dev/config/
    export default defineConfig(({mode}) => {
    
        const __fileName = fileURLToPath(import.meta.url)
        const __dirname = dirname(__fileName)
        const env = loadEnv(mode, __dirname, '')
        const proxyTarget = env.VITE_DEV_PROXY_TARGET || 'http://localhost:3000'
    
        return {
            plugins: [
                react(),
                tailwindcss(),
                babel({ presets: [reactCompilerPreset()] }),
            ],
            resolve: {
                alias: {
                    '@': path.resolve(__dirname, './src'),
                },
            },
            server: {
                proxy: {
                    '/api': {
                        target: proxyTarget,
                        changeOrigin: true
                    }
                }
            }
        }
    })
```
